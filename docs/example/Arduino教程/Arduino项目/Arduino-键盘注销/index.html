    Arduino - 键盘注销#

    更新于 2024/6/19 19:01:29

    当 ARDUINO UNO 上的引脚 2 接地时，此示例使用键盘库将您从计算机上的用户会话中注销。草图模拟同时按顺序按下两个或三个键，并在短暂延迟后释放它们。

    警告 − 当您使用 Keyboard.print() 命令时，Arduino 会接管您计算机的键盘。为确保您在运行带有此功能的草图时不会失去对计算机的控制，请在调用 Keyboard.print() 之前设置可靠的控制系统。此草图设计为仅在引脚接地后发送键盘命令。

    草图#

    在计算机上打开 Arduino IDE 软件。使用 Arduino 语言编码将控制您的电路。单击新建打开一个新的草图文件。

    对于此示例，您需要使用 Arduino IDE 1.6.7

    注意 −您必须将键盘库包含在 Arduino 库文件中。将键盘库文件复制并粘贴到名为"libraries"（突出显示）的文件中，如以下屏幕截图所示。

    Arduino 代码#

    /*
键盘注销
此草图演示了键盘库。
当您将引脚 2 接地时，它会执行注销。
它使用键盘组合来执行此操作，如下所示：
在 Windows 上，按 CTRL-ALT-DEL，然后按 ALT-l
在 Ubuntu 上，按 CTRL-ALT-DEL 和 ENTER
在 OSX 上，按 CMD-SHIFT-q
唤醒：空格键。
电路：
* Arduino Leonardo 或 Micro
* 电线将 D2 接地。
*/

#define OSX 0
#define WINDOWS 1
#define UBUNTU 2

#include "Keyboard.h"

// 更改此设置以匹配您的平台：
int platform = WINDOWS;

void setup() {
    // 将引脚 2 设为输入并打开
    // 上拉电阻，使其升高，除非
    // 接地：
   
   pinMode(2, INPUT_PULLUP);
   Keyboard.begin();
}

void loop() {
   while (digitalRead(2) == HIGH) {
      // 直到引脚 2 变为低电平，才执行任何操作
      delay(500);
   }
   
   delay(1000);
   
   switch (platform) {
      case OSX:
      Keyboard.press(KEY_LEFT_GUI);
	  
      // Shift-Q logs out:
      Keyboard.press(KEY_LEFT_SHIFT);
      Keyboard.press('Q');
      delay(100);
	  
      // enter:
      Keyboard.write(KEY_RETURN);
      break;
	  
      case WINDOWS:
      // CTRL-ALT-DEL:
      Keyboard.press(KEY_LEFT_CTRL);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press(KEY_DELETE);
      delay(100);
      Keyboard.releaseAll();
	  
      //ALT-l:
      delay(2000);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press('l');
      Keyboard.releaseAll();
      break;
	  
      case UBUNTU:
      // CTRL-ALT-DEL:
      Keyboard.press(KEY_LEFT_CTRL);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press(KEY_DELETE);
	  
      delay(1000);
      Keyboard.releaseAll();
	  
      // Enter to confirm logout:
      Keyboard.write(KEY_RETURN);
      break;
   }
   
   // do nothing:
   while (true);
}

Keyboard.releaseAll();

   // enter:
      Keyboard.write(KEY_RETURN);
      break;
      case WINDOWS:
	  
   // CTRL-ALT-DEL:
      Keyboard.press(KEY_LEFT_CTRL);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press(KEY_DELETE);
      delay(100);
      Keyboard.releaseAll();
	  
   //ALT-l:
      delay(2000);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press('l');
      Keyboard.releaseAll();
      break;
	  
   case UBUNTU:
      // CTRL-ALT-DEL:
      Keyboard.press(KEY_LEFT_CTRL);
      Keyboard.press(KEY_LEFT_ALT);
      Keyboard.press(KEY_DELETE);
      delay(1000);
      Keyboard.releaseAll();
	  
      // Enter to confirm logout:
      Keyboard.write(KEY_RETURN);
      break;
   }
   
// 不执行任何操作：
while (true);
}

    需要注意的代码#

    在将程序上传到主板之前，请确保将当前使用的正确操作系统分配给平台变量。

    在草图运行时，按下按钮将把引脚 2 连接到地，主板将向 USB 连接的 PC 发送注销序列。
